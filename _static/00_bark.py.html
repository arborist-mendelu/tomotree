import sys
sys.path.append('../src')
import tomotree as tomo
import pandas as pd 
import matplotlib.pyplot as plt
from matplotlib.ticker import FuncFormatter

DIRECTORY = '../streamlit_data/Jičín_05/'
NX = 61
NY = 61
nodes_file = DIRECTORY + 'nodes-05.csv'
speeds_file = DIRECTORY + 'speeds-05.csv'

df_nodes = pd.read_csv(nodes_file, header=None)
df_nodes = df_nodes - df_nodes.mean(axis=0)
df_speeds = pd.read_csv(speeds_file, header=None)

for bark in [0,0.05,0.1]:
    t = tomo.Tomo(
        df_nodes.values, 
        df_speeds.values, 
        nx=NX, 
        ny=NY, 
        symmetric=True, 
        bark=bark,
        )
    t.do_ERSM_method()
    t.optimize_back_tomogram()
    sound_wood_speed = t.estimate_sound_wood_speed()    
    fig, ax = plt.subplots()
    t.plot(
            rays=False, 
            grid=True, 
            ax=ax,  
            speed_max=sound_wood_speed, 
            speed_min=(sound_wood_speed)*0.3
            )
    cm_formatter = FuncFormatter(lambda x, pos: f"{x*100:.0f}")
    ax.grid(True)
    ax.axis('on')
    ax.xaxis.set_major_formatter(cm_formatter)
    ax.yaxis.set_major_formatter(cm_formatter)
    ax.set_xlabel('x / cm')
    ax.set_ylabel('y / cm')
    plt.suptitle(f"Bark thickness: {bark*100:.0f} cm, Sound speed: {sound_wood_speed:.0f} m/s")
    plt.tight_layout()
